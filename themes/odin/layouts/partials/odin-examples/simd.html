package main import "core:simd" // Vectorized check that a utf8 buffer is all ASCII is_ascii :: proc(utf8: []#simd[16]u8) -> bool { acc: #simd[16]u8 for chunk in utf8 { acc |= chunk // fold every set of code units into the result } return simd.reduce_max(acc) < 0x80 // every lane in the ASCII range? }
